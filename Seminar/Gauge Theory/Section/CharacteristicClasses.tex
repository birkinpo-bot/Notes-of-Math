\section{Characteristic Classes}
Let $\pi:E\to M$ be a rank $r$ vector bundle, 
$\nabla$ be a connection on $E$.
Its curvature $R^\nabla := \nabla^2$ is an $\End(E)$-valued 2-form on $M$. 
Under a local trivialization $\{(U_\alpha,\psi_\alpha)\}$ of $E$, 
$\nabla$ can be represented by an $r\times r$ matrix of 1-forms $\omega_\alpha$, 
and $R^\nabla$ can be represented by an $r\times r$ matrix of 2-forms $\Omega_\alpha$.  

The main goal of this section is the following: 
\begin{enumerate}
    \item An invariant polynomial $\Phi$  on $\mathfrak{gl}(r,\mathbb{R})$ 
    determines a global form $\Phi(R^\nabla)$ on $M$.
    \item The global form $\Phi(R^\nabla)$ is closed.
    \item The class $[\Phi(R^\nabla)]$ is independent of the choice of $\nabla$, 
    thus defines an invariant of $E$.
    \item \emph{Chern-Weil homomorphism}: $\mathrm{Inv}(\mathfrak{gl}(r,\mathbb{R}))\to H_{\mathrm{dR}}^{*}(M)$, $\Phi\mapsto[\Phi(R^\nabla)]$. 
    \item Examples: \emph{Chern classes}, \emph{Pontryagin classes} and \emph{Euler classes}. 
    \item Generalization to principal $G$-bundles: $\mathrm{Inv}(\mathfrak{g})\to H_{\mathrm{dR}}^{*}(M)$.
\end{enumerate}

\subsection{Chern-Weil Theory}
Repeat the settings above: 
Let $\pi:E\to M$ be a rank $r$ vector bundle, 
$\nabla$ a connection on $E$.
The curvature $R^\nabla$ is an $\End(E)$-valued 2-form on $M$. 
Under a local trivialization $\{(U_\alpha,\psi_\alpha)\}$ of $E$, 
$\nabla$ can be represented by an $r\times r$ matrix of 1-forms $\omega_\alpha$, 
and $R^\nabla$ can be represented by an $r\times r$ matrix of 2-forms $\Omega_\alpha$. 

Recall that the curvature form $\Omega_\alpha$ is defined by 
\begin{equation*}
    \Omega_\alpha = d\omega_\alpha + \omega_\alpha\wedge\omega_\alpha.
\end{equation*}
It satisfies the second Bianchi identity
\begin{equation*}
    d\Omega_\alpha = \Omega_\alpha\wedge\omega_\alpha - \omega_\alpha\wedge\Omega_\alpha,
\end{equation*}
and the general Bianchi identity
\begin{equation*}
    d\Omega_\alpha^k = \Omega_\alpha^k\wedge\omega_\alpha - \omega_\alpha\wedge\Omega_\alpha^k.
\end{equation*}
The transformation formula for $\omega_\alpha$ and $\Omega_\alpha$ is given by
\begin{gather*}
\omega_\beta = g_{\alpha\beta}^{-1}\omega_\alpha g_{\alpha\beta} 
+ g_{\alpha\beta}^{-1}dg_{\alpha\beta}, \qquad 
\Omega_\beta = g_{\alpha\beta}^{-1}\Omega_\alpha g_{\alpha\beta}.
\end{gather*}
where $g_{\alpha\beta}$ is the transition function. 

% Since $\Omega_\alpha$ transforms by conjugation, 
% if we apply a function $\Phi:\mathbb{R}^{r\times r}\to\mathbb{R}$ that is invariant under conjugation, 
% then $\Phi(\Omega_\alpha)$ and $\Phi(\Omega_\beta)$ will coincide on $U_\alpha\cap U_\beta$, 
% thus determine a global form on $M$. 

\begin{definition}
    Let $X = (x^i_j)$ be an $r\times r$ matrix with indeterminate entries. 
    A polynomial $\Phi(X)$ on $\mathfrak{gl}(r,\mathbb{R})=\mathbb{R}^{r\times r}$ is a polynomial in the variables $x^i_j$. 
    A polynomial $\Phi(X)$ on $\mathfrak{gl}(r,\mathbb{R})$ is \textbf{$\mathrm{Ad}$-invariant (or simply invariant)} if
    \begin{equation}\label{eq:invariant_polynomial}
        \Phi(A^{-1}XA) = \Phi(X), \qquad \forall\, A\in\mathrm{GL}(r,\mathbb{R}).
    \end{equation}
\end{definition}
\begin{remark}
    If $X$ is a matrix of real numbers, then $\Phi(X)$ is a real number. 
    If $X$ is a matrix of 2-forms, then $\Phi(X)$ is also a differential form.
\end{remark}
\begin{fact}
    If \eqref{eq:invariant_polynomial} holds for all $r\times r$ matrices $X$ of real numbers, 
    then it also holds when $X$ is a matrix of indeterminate. 
    Thus $\Tr(X)$ and $\det(X)$ are invariant polynomials.
\end{fact}
\begin{example}
    Let $\det(\lambda I + X) = \lambda^r + f_1(X)\lambda^{r-1} + \cdots + f_r(X)$, 
    then $f_k(X)$ is an invariant polynomial of degree $k$. 
    In fact, $(-1)^kf_k(X)$ is the $k$-th coefficient of the characteristic polynomial of $X$. 
\end{example}
\begin{example}
    Let $\Sigma_k(X)=\Tr(X^k)$ be the trace of $X^k$, then $\Sigma_k(X)$ is an invariant polynomial of degree $k$.
\end{example}

\begin{proposition}\label{prop:global_form}
    Let $\Phi(X)$ be an invariant polynomial on $\mathfrak{gl}(r,\mathbb{R})$. 
    Then $\Phi(\Omega_\alpha)$ and $\Phi(\Omega_\beta)$ coincide on $U_\alpha\cap U_\beta$, 
    \textcolor{red}{thus determine a global form $\Phi(R^\nabla)$ on $M$.}
\end{proposition}
\begin{proof}
    Since $\Omega_\beta = g_{\alpha\beta}^{-1}\Omega_\alpha g_{\alpha\beta}$, 
    we have 
    \begin{equation*}
        \Phi(\Omega_\beta) = \Phi(g_{\alpha\beta}^{-1}\Omega_\alpha g_{\alpha\beta}) = \Phi(\Omega_\alpha),
        \quad \text{on } U_\alpha\cap U_\beta. \qedhere
    \end{equation*}
\end{proof}

\begin{proposition}\label{prop:closed}
    \textcolor{red}{The global form $\Phi(R^\nabla)$ is closed}.
\end{proposition}
To prove this proposition, we need the following fact about invariant polynomials.
\begin{fact}
    The algebra of invariant polynomials on $\mathfrak{gl}(r,\mathbb{R})$ is 
    the polynomial algebra generated by $\Sigma_1,\dots,\Sigma_r$ 
    (or equivalently by $f_1,\dots,f_r$), i.e., 
    \begin{equation*}
        \mathrm{Inv}(\mathfrak{gl}(r,\mathbb{R})) 
        = \mathbb{R}[\Sigma_1,\dots,\Sigma_r] 
        = \mathbb{R}[f_1,\dots,f_r].
    \end{equation*}
\end{fact}
For example, suppose $\Phi$ is a homogeneous invariant polynomial of degree $3$ on $\mathfrak{gl}(3,\mathbb{R})$, 
then $\Phi$ must be a linear combination of $\Sigma_1^3$, $\Sigma_1\Sigma_2$ and $\Sigma_3$, i.e.,
\begin{equation*}
    \Phi(X) = a\Sigma_1(X)^3 + b\Sigma_1(X)\Sigma_2(X) + c\Sigma_3(X), 
    \qquad a,b,c\in\mathbb{R}.
\end{equation*}

\begin{proof}[Proof of Proposition \ref{prop:closed}]
    Since $\Phi$ is a polynomial in $\Sigma_1,\dots,\Sigma_r$, 
    it suffices to show that $\md\Tr(R^\nabla)=0$ for all $k$. 
    We have 
    \begin{equation*}
        \md\Tr(\Omega_\alpha^k) = \Tr(\md\Omega_\alpha^k)
        = \Tr(\Omega_\alpha^k\wedge\omega_\alpha - \omega_\alpha\wedge\Omega_\alpha^k) 
        = 0. \qedhere
    \end{equation*}
\end{proof}

\begin{proposition}\label{prop:independent}
    \textcolor{red}{The class $[\Phi(R^\nabla)]$ is independent of the choice of $\nabla$.}
    It is a {\rm(}differential{\rm)} topological invariant of $E$.
\end{proposition}
\begin{proof}
    For the same reason as above, it suffices to show that $[\Tr(R^\nabla)]$ is independent of the choice of $\nabla$. 
    Let $\nabla_0, \nabla_1$ be two connections on $E$, then $\nabla_t = (1-t)\nabla_0 + t\nabla_1$ is a family of connections on $E$, 
    whose connection form is given by 
    \begin{equation*}
        \omega_t = (1-t)\omega_0 + t\omega_1,
    \end{equation*}
    and curvature form is 
    \begin{equation*}
        \qquad \Omega_t = \md\omega_t + \omega_t\wedge\omega_t.
    \end{equation*}
    So 
    \begin{equation*}
        \dif{\Omega_t}{t} = \dif{}{t}(\md\omega_t + \omega_t\wedge\omega_t)
        = \dif{}{t}(\md\omega_t) 
        + \dif{\omega_t}{t}\wedge\omega_t 
        + \omega_t\wedge\dif{\omega_t}{t} 
    \end{equation*}
    Then 
    \begin{align*}
        \dif{}{t}\Tr(\Omega_t^k) 
        &= \Tr\left(\dif{}{t}\Omega_t^k\right) 
        = \Tr\left(\dif{\Omega_t}{t}\wedge\Omega_t^{k-1} + \cdots + \Omega_t^{k-1}\wedge\dif{\Omega_t}{t}\right) 
        = k\Tr\left(\dif{\Omega_t}{t}\wedge\Omega_t^{k-1}\right) \\
        &= k\Tr\left(\left(\md\dif{\omega_t}{t} + \dif{\omega_t}{t}\wedge\omega_t + \omega_t\wedge\dif{\omega_t}{t}\right)
        \wedge\Omega_t^{k-1}\right) \\ 
        &= k\Tr\left(\md\left(\dif{\omega_t}{t}\right)\wedge\Omega_t^{k-1} 
        - \dif{\omega_t}{t}\wedge(\Omega_t^{k-1}\wedge\omega_t - \omega_t\wedge\Omega_t^{k-1})\right) \\ 
        &= k\Tr\left(\md\left(\dif{\omega_t}{t}\right)\wedge\Omega_t^{k-1} - \dif{\omega_t}{t}\wedge\md\Omega_t^{k-1}\right) \\ 
        &= k\Tr\left(\md\left(\dif{\omega_t}{t}\wedge\Omega_t^{k-1}\right)\right) \\ 
        &= \md\, k\Tr\left(\dif{\omega_t}{t}\wedge\Omega_t^{k-1}\right).
    \end{align*}
    Therefore, 
    \begin{align}\label{eq:transgression}
        \Tr(\Omega_1^k) - \Tr(\Omega_0^k) 
        &= \int_0^1 \dif{}{t}\Tr(\Omega_t^k) \md t 
        = \int_0^1 \md\, k\Tr\left(\dif{\omega_t}{t}\wedge\Omega_t^{k-1}\right) \md t \notag\\
        &= \md\int_0^1 k\Tr\left(\dif{\omega_t}{t}\wedge\Omega_t^{k-1}\right) \md t. 
    \end{align}
    Notice that 
    \begin{equation*}
        \omega_{t,\beta} = g_{\alpha\beta}^{-1}\omega_{t,\alpha} g_{\alpha\beta} 
        + g_{\alpha\beta}^{-1}dg_{\alpha\beta} \implies 
        \dif{\omega_{t,\beta}}{t} = g_{\alpha\beta}^{-1}\dif{\omega_{t,\alpha}}{t} g_{\alpha\beta},\quad
        \Omega_{t,\beta} = g_{\alpha\beta}^{-1}\Omega_{t,\alpha} g_{\alpha\beta}.
    \end{equation*} 
    So $\Tr\left(\dif{\omega_t}{t}\wedge\Omega_t^{k-1}\right)$
    defines a global form on $M$. By \eqref{eq:transgression}, 
    we have $[\Tr(\Omega_1^k)] = [\Tr(\Omega_0^k)]$. 

    For a general invariant polynomial $\Phi$, 
    % since $\Phi$ is a polynomial in $\Sigma_1,\dots,\Sigma_r$, 
    we can express $\Phi$ as a linear combination of monomials $\Sigma_1^{k_1}\cdots\Sigma_r^{k_r}$. 
    Then $\Phi(\Omega_t)$ is a linear combination of $\Sigma_1^{k_1}(\Omega_t)\wedge\cdots\wedge\Sigma_r^{k_r}(\Omega_t)$. 
    Thus $[\Phi(\Omega_1)] = [\Phi(\Omega_t)] = [\Phi(\Omega_0)]$. 
\end{proof}

\begin{definition}
    By above propositions,
    we define the \textbf{Chern-Weil homomorphism} as 
    \begin{align*}
        c_E:\mathrm{Inv}(\mathfrak{gl}(r,\mathbb{R})) &\to H_{\mathrm{dR}}^{*}(M), \\
        \Phi &\mapsto [\Phi(R^\nabla)].
    \end{align*}
    This map sends $\Phi(X)\Psi(X)$ to $[\Phi(R^\nabla)\wedge\Psi(R^\nabla)] = [\Phi(R^\nabla)]\cdot[\Psi(R^\nabla)]$, 
    thus is an algebra homomorphism. This map does not depend on the choice of $\nabla$. 
\end{definition}

\begin{definition} %[Characteristic Classes]
    A \textbf{characteristic class} of a vector bundle $\pi:E\to M$ is a map 
    \begin{equation*}
        c:\mathrm{Vect}_k(M)\to H_{\mathrm{dR}}^{*}(M), \qquad E\mapsto c(E),
    \end{equation*} 
    such that if $f:N\to M$ is a smooth map, then 
    \begin{equation*}
        c(f^*E) = f^*c(E).
    \end{equation*}
\end{definition}

In functional language, a \textbf{characteristic class} is a natural transformation from 
the functor $\mathrm{Vect}_k(-)$ to $H_{\mathrm{dR}}^{*}(-)$. 
The naturality condition can be visualized by the following commutative diagram: 
\begin{center}
    \begin{tikzcd}
        \mathrm{Vect}_k(M) \arrow[r, "c"] & H_{\mathrm{dR}}^{*}(M) \\
        \mathrm{Vect}_k(N) \arrow[u, "f^*"] \arrow[r, "c"] & H_{\mathrm{dR}}^{*}(N) \arrow[u, "f^*"]
    \end{tikzcd}
\end{center}

\subsubsection*{Naturality of Chern-Weil homomorphism}
Suppose $\pi:E\to M$ is a rank $r$ vector bundle, $\nabla$ a connection on $E$.
Let $f:N\to M$ be a smooth map, it induces a pullback bundle $f^*E\to N$ and a pullback connection $f^*\nabla$ on $f^*E$ 
\begin{equation*}
    (f^*\nabla)(f^*s) := f^*(\nabla s), \qquad \forall\, s\in\Gamma(E).
\end{equation*}
Suppose $\omega$ is the connection form of $\nabla$, then $f^*\omega$ is the connection form of $f^*\nabla$.
The curvature form of $f^*\nabla$ is therefore $f^*(\md\omega + \omega\wedge\omega) = f^*\Omega$.
For any invariant polynomial $\Phi$, we have
\begin{equation*}
    \Phi(f^*\Omega) = f^*\Phi(\Omega) \implies
    c_{f^*E}(\Phi) = [\Phi(f^*\Omega)] = f^*[\Phi(\Omega)] = f^*c_E(\Phi).
\end{equation*}
which shows the naturality of Chern-Weil homomorphism.

\subsection{Examples of Characteristic Classes}
We have introduced a powerful mechinery to construct characteristic classes of vector bundles. 
In this subsection, we will give a few examples of characteristic classes. 

\subsubsection{Pontryagin Classes}
In this subsection, we only consider real vector bundles. 
Recall that 
\begin{equation*}
    \det(\lambda I + X) = \lambda^r + f_1(X)\lambda^{r-1} + \cdots + f_r(X),
\end{equation*}
in particular, $f_1(X) = \Tr(X)$. If $X$ is a skew-symmetric matrix, then $f_1(X) = 0$. 
In general we have the following proposition.

\begin{proposition}
    If $\Omega$ is skew-symmetric, then $\Tr(\Omega^k) = 0$ for all odd $k$. 
    Moreover, $\Phi(\Omega) = 0$ for all invariant polynomials $\Phi$ of odd degree. 
\end{proposition}
\begin{proof}
    Since $\Omega$ is skew-symmetric, 
    \begin{equation*}
        (\Omega\wedge\cdots\wedge\Omega)^T = (\Omega^T)\wedge\cdots\wedge(\Omega^T) 
        = (-\Omega)\wedge\cdots\wedge(-\Omega) = (-1)^k\Omega\wedge\cdots\wedge\Omega.
    \end{equation*}
    So $\Omega^k$ is skew-symmetric for odd $k$, and $\Tr(\Omega^k) = 0$. 

    For a general invariant polynomial $\Phi$ of odd degree, 
    since $\Phi$ is a linear combination of monomials $\Sigma_1^{k_1}\cdots\Sigma_r^{k_r}$ with $k_1 + 2k_2 + \cdots + rk_r$ odd,
    at least one of $k_1,k_3,\dots$ is odd, thus $\Phi(\Omega)$ must be zero.
\end{proof}

Since $[\Phi(R^\nabla)]$ is independent of the choice of $\nabla$, 
we can choose a special connection $\nabla$ such that $R^\nabla$ is skew-symmetric. 
To do this, we choose a Riemannian metric on $E$, then there exists connections on $E$ that are compatible with the metric. 
Under a local orthonormal frame, the connection form $\omega$ is skew-symmetric, so as the curvature form $\Omega$. 
By the above proposition, $[\Phi(R^\nabla)] = 0$ for all invariant polynomials $\Phi$ of odd degree.
\begin{corollary}
    For a real vector bundle $E$, the ring of characteristic classes of $E$ has two sets of generators:
    \begin{equation*}
        [\Tr(\Omega^2)],\, [\Tr(\Omega^4)],\, [\Tr(\Omega^6)], \dots; 
    \end{equation*}
    and
    \begin{equation*}
        [f_2(\Omega)],\, [f_4(\Omega)],\, [f_6(\Omega)], \dots.
    \end{equation*}
\end{corollary}
\begin{definition}
    The $k$-th \textbf{Pontryagin class} of a real vector bundle $E$ is defined to be 
    \begin{equation*}
        p_k(E) := (-1)^k\left[f_{2k}\left(\frac{1}{2\pi}R^\nabla\right)\right] = \left[f_{2k}\left(\frac{\mi}{2\pi}R^\nabla\right)\right] \in H^{4k}(M).
    \end{equation*}
\end{definition}
\begin{remark}
    The factor $\mi$ is for the sign convention. The factor $\frac{1}{2\pi}$ is for the integrality of Pontryagin classes. 
    A class $\alpha\in H^{q}(M)$ is called \textbf{integral} 
    if $\int_S \alpha\in\mathbb{Z}$ for all $q$-dimensional closed oriented submanifolds $S$ of $M$.
\end{remark}
By definition, 
\begin{align*}
    \det\left(\lambda I + \frac{\mi}{2\pi}R^\nabla\right) 
    &= \lambda^r + f_1\left(\frac{\mi}{2\pi}R^\nabla\right)\lambda^{r-1} + \cdots + f_r\left(\frac{\mi}{2\pi}R^\nabla\right) \\
    &= \lambda^r + p_1(E)\lambda^{r-2} + \cdots + p_{[r/2]}(E)\lambda^{r-2[r/2]}.
\end{align*}
Apply $\lambda = 1$, 
\begin{equation*}
    \det\left(I + \frac{\mi}{2\pi}R^\nabla\right) = 1 + p_1(E) + \cdots + p_{[r/2]}(E) =: p(E).
\end{equation*} 
We call $p(E)$ the \textbf{total Pontryagin class} of $E$.
\begin{proposition}[Whitney product formula]
    For two real vector bundles $E$ and $E'$, we have 
    \begin{equation*}
        p(E\oplus E') = p(E)\cdot p(E').
    \end{equation*}
\end{proposition}
\begin{proof}
    Let $\nabla$ and $\nabla'$ be connections on $E$ and $E'$, then $\nabla\oplus\nabla'$ is a connection on $E\oplus E'$. 
    Under a suitable local trivialization, the curvature form of $\nabla\oplus\nabla'$ is given by 
    \begin{equation*}
        F_{\nabla\oplus\nabla'} = 
        \begin{pmatrix}
            R^\nabla & 0 \\
            0 & R^{\nabla'}
        \end{pmatrix} \implies 
        \Omega_{E\oplus E'} = 
        \begin{pmatrix}
            \Omega & 0 \\
            0 & \Omega'
        \end{pmatrix}.
    \end{equation*}
    So 
    \begin{align*}
        p(E\oplus E')
        &= \det\left(I + \frac{\mi}{2\pi}\Omega_{E\oplus E'}\right) \\ 
        &= \det\begin{pmatrix}
            I + \frac{\mi}{2\pi}\Omega &  \\
             & I + \frac{\mi}{2\pi}\Omega'
        \end{pmatrix} \\
        &= \det\left(I + \frac{\mi}{2\pi}\Omega\right)\cdot\det\left(I + \frac{\mi}{2\pi}\Omega'\right) \\
        &= p(E)\cdot p(E'). \qedhere
    \end{align*}
\end{proof}

\subsubsection{Euler Classes}
If in addition $E$ is orientable and we choose an orientation of $E$. 
Then we can make the local orthonormal frame $e = [e_1,\dots,e_r]$ of $E$ to be compatible with the orientation,  
so the transition function between two frames is in $\mathrm{SO}(r)$ and the connection form $\omega$ is skew-symmetric. 
This inspires us to consider the $\mathrm{SO}(r)$-invariant polynomial on $\mathfrak{so}(r)$. 

\begin{fact}
    For $r$ odd, $\mathrm{Inv}(\mathfrak{so}(r))$ is generated by $\Sigma_1,\dots,\Sigma_r$ (or equivalently by $f_1,\dots,f_r$).
    For $r$ even, $\mathrm{Inv}(\mathfrak{so}(r))$ has one more generator $\Pf$, the \textbf{Pfaffian}. 
    That is
    \begin{gather*}
        \mathrm{Inv}(\mathfrak{so}(2m+1)) = \mathbb{R}[\Sigma_1,\dots,\Sigma_{2m+1}] = \mathbb{R}[f_1,\dots,f_{2m+1}], \\
        \mathrm{Inv}(\mathfrak{so}(2m)) = \mathbb{R}[\Sigma_1,\dots,\Sigma_{2m},\Pf\,] = \mathbb{R}[f_1,\dots,f_{2m},\Pf\,].
    \end{gather*}
\end{fact}

\subsubsection*{The Pfaffian of skew-symmetric matrices}
Let's begin with some examples. For a $2\times 2$ skew-symmetric matrix 
\begin{equation*}
    X = 
    \begin{pmatrix}
        0 & a \\
        -a & 0
    \end{pmatrix},
\end{equation*}
we have 
\begin{equation*}
    \det(X) = a^2.
\end{equation*}
For a $4\times 4$ skew-symmetric matrix
\begin{equation*}
    X = 
    \begin{pmatrix}
        0 & a & b & c \\
        -a & 0 & d & e \\
        -b & -d & 0 & f \\
        -c & -e & -f & 0
    \end{pmatrix},
\end{equation*}
after a direct calculation, 
\begin{equation*}
    \det(X) = (af - be + cd)^2.
\end{equation*}
One can observe that both determinants are perfect squares. 
This phenomenon holds for all skew-symmetric matrices:
\begin{fact}
    For any $2m\times 2m$ skew-symmetric matrix $X=(x^i_j)$, 
    $\det(X)$ is a perfect square in $\mathbb{Z}[x^i_j]$. 
    The square root of $\det(X)$ is called the \textbf{Pfaffian} of $X$, 
    denoted by $\Pf(X)$. 
\end{fact}
\begin{remark}
    Notice that the square roots of $\det(X)$ are $\pm\Pf(X)$. 
    To determine the sign, we require $\Pf(J_{2m}) = 1$, where 
    \begin{equation*}
        J_{2m} = 
        \begin{pmatrix}
            0 & 1 &  &  & \\
            -1 & 0 &  &  & \\
             &  & \ddots &  & \\
             &  &  &  0 & 1 \\
             & &  & -1 & 0
        \end{pmatrix}.
    \end{equation*}
\end{remark}
\begin{fact}
    Let $A=(a^i_j)$ and $X=(x^i_j)$ be two $2m\times 2m$ matrices with $X$ skew-symmetric, then 
    \begin{equation*}
        \Pf(A^T X A) = \det(A)\Pf(X).
    \end{equation*}
    If $A\in\mathrm{SO}(2m)$, then $\Pf(A^T X A) = \Pf(X)$, 
    so $\Pf$ is an invariant polynomial on $\mathfrak{so}(2m)$.
\end{fact}
\begin{definition}
    For a $2m$-dimensional oriented real vector bundle $E\to M$, 
    choose a metric on $E$ and a compatible connection $\nabla$, 
    the \textbf{Euler class} of $E$ is defined to be
    \begin{equation*}
        e(E) := \left[\Pf\left(\frac{1}{2\pi}R^\nabla\right)\right] \in H^{2m}(M).
    \end{equation*} 
\end{definition}
\begin{remark}
    The Euler class is a characteristic class of oriented real vector bundles. 
    It depends on the orientation: 
    if we reverse the orientation, then $e(E)$ changes to $-e(E)$.
\end{remark}
\begin{remark}
    The definition of Euler class requires a metric and a compatible connection, 
    but the resulting class does not depend on the choice of metric and connection:
    \begin{itemize}[leftmargin=*]
        \item Fixing a metric $g$ on $E$, any two metric-compatible connections 
        $\nabla_0$ and $\nabla_1$ define the same Euler class. 
        \item Given any two metrics $g_0$ and $g_1$ on $E$, 
        we can choose a family of connections $\nabla_t$ such that 
        $\nabla_0$ (resp. $\nabla_1$) is compatible with $g_0$ (resp. $g_1$). 
        The whole family $\nabla_t$ defines the same Euler class.
    \end{itemize} 
\end{remark}
\begin{theorem}[Gauss-Bonnet-Chern theorem]
    Let $M$ be a closed oriented Riemannian manifold of dimension $2m$, 
    $\nabla$ be a metric-compatible connection on $TM$, then
    \begin{equation*}
        \int_M e(TM) = \int_M \Pf\left(\frac{1}{2\pi}R^\nabla\right) = \chi(M),
    \end{equation*}
    where $\chi(M)$ is the Euler characteristic of $M$.    
\end{theorem}
\begin{example}
    When $M$ is a closed oriented surface, the above theorem reduces to the classical Gauss-Bonnet theorem: 
    let $e_1,e_2$ be a local orthonormal frame of $TM$, $\theta^1,\theta^2$ be the dual coframe, 
    then the connection and curvature forms are
    \begin{equation*}
        \omega = 
        \begin{pmatrix}
            0 & \omega^1_2 \\
            -\omega^1_2 & 0
        \end{pmatrix}, \qquad
        \Omega = \md\omega + \omega\wedge\omega = 
        \begin{pmatrix}
            0 & \md\omega^1_2 \\
            -\md\omega^1_2 & 0
        \end{pmatrix},
    \end{equation*}
    and $\md\omega^1_2 = K\theta^1\wedge\theta^2$, where $K$ is the Gaussian curvature of $M$. 
    So
    \begin{equation*}
        \Pf\left(\Omega\right) = \md\omega^1_2 = K\theta^1\wedge\theta^2.
    \end{equation*}
    Therefore,
    \begin{equation*}
        \int_M e(TM) = \frac{1}{2\pi}\int_M K\theta^1\wedge\theta^2 = \chi(M) = 2 - 2g(M).
    \end{equation*}
\end{example}

\subsubsection{Chern Classes}
For a complex vector bundle $E\to M$ with $\rk_{\mC}(E)=r$, 
we can choose a local trivialization $\{(U_\alpha,\psi_\alpha)\}$ of $E$ 
such that the transition function $g_{\alpha\beta}\in\mathrm{GL}(r,\mathbb{C})$. 
Then any complex polynomial $\Phi$ on $\mathfrak{gl}(r,\mathbb{C})$ 
invariant under conjugation by $\mathrm{GL}(r,\mathbb{C})$ 
defines a characteristic class $[\Phi(R^\nabla)]$ which is independent of the choice of $\nabla$. 

% Taking $\Phi(X) = \det(\lambda I + \frac{i}{2\pi}X)$, we obtain the \textbf{Chern classes} of $E$:
\begin{definition}
    The $k$-th \textbf{Chern class} of a complex vector bundle $E$ is defined to be 
    \begin{equation*}
        c_k(E) := \left[f_k\left(\frac{\mi}{2\pi}R^\nabla\right)\right] \in H^{2k}(M).
    \end{equation*}
    The \textbf{total Chern class} of $E$ is defined to be
    \begin{equation*}
        c(E) := \det\left(I + \frac{\mi}{2\pi}R^\nabla\right) = 1 + c_1(E) + \cdots + c_r(E).
    \end{equation*}
\end{definition}
\begin{proposition}
    Chern classes satisfy the naturality property and the Whitney product formula, i.e.,
    \begin{equation*}
        c(f^*E) = f^*c(E), \qquad c(E\oplus E') = c(E)\cdot c(E').
    \end{equation*}
\end{proposition}
\begin{remark}
    If we choose a Hermitian metric on $E$ and a compatible connection $\nabla$, 
    then the curvature form $R^\nabla$ is skew-Hermitian, 
    so $\frac{\mi}{2\pi}R^\nabla$ is Hermitian. Therefore, 
    $\Sigma_k\left(\frac{\mi}{2\pi}R^\nabla\right) = \Tr\left(\left(\frac{\mi}{2\pi}R^\nabla\right)^k\right)$ 
    is a real form, and $c_k(E)$ is a real cohomology class. 
    In fact, $c_k(E)$ is an integral cohomology class, i.e., $c_k(E)\in H^{2k}(M;\mathbb{Z})$.
\end{remark}

\begin{proposition}
    Let $E$ be a complex vector bundle, $\rk_{\mC}(E) = r$, 
    $E_\mR$ be the underlying real vector bundle of $E$. 
    $E_\mR$ is oriented and if $e = [e_1,\cdots,e_r]$ is a local frame of $E$, 
    then $e_\mR = [e_1,\mi e_1,\cdots,e_r,\mi e_r]$ is a local oriented frame of $E_\mR$. 
    \textcolor{blue}{We set the orientation determined by $e_\mR$ to be the orientation of $E_\mR$,}
    then the top Chern class $c_r(E)$ is equal to the Euler class $e(E_\mR)$. 
\end{proposition}
\begin{proof}
    Choose a Hermitian metric $\langle \cdot, \cdot \rangle$ on $E$ and a compatible connection $\nabla$, 
    then the curvature $R^\nabla$ is skew-Hermitian. One can define a Riemannian metric on $E_\mR$ by 
    \begin{equation*}
        \langle u,v\rangle_\mR := \mathrm{Re}\langle u,v\rangle\qquad\forall\,u,v\in E_\mR.
    \end{equation*}
    Then
    \begin{equation*}
        X\langle u,v\rangle_\mR = \mathrm{Re}\Big(X\langle u,v\rangle\Big) 
        = \mathrm{Re}\Big(\langle\nabla_X u,v\rangle + \langle u,\nabla_X v\rangle\Big)
         = \langle\nabla_X u,v\rangle_\mR + \langle u,\nabla_X v\rangle_\mR.
    \end{equation*}
    So $\nabla$ is compatible with $\langle \cdot, \cdot \rangle_\mR$, 
    and the curvature $(R^{\nabla})_\mR$ is skew-symmetric. 
    
    Under a local unitary frame $e = [e_1,\dots,e_r]$ of $E$, 
    the curvature form $\Omega$ is skew-Hermitian, and so $\mi\Omega$ is Hermitian.
    The frame $e_\mR = [e_1,\mi e_1,\cdots,e_r,\mi e_r]$ is a local orthonormal frame of $E_\mR$, 
    and the curvature form under $e_\mR$ is skew-symmetric. 

    If $\Omega = (z^i_j) = (x^i_j + \mi y^i_j)$ with $x^i_j,y^i_j\in\mathbb{R}$, 
    then $\Omega = A + \mi B$ with $A = (x^i_j)$ and $B = (y^i_j)$. 
    Then the curvature form under $e_\mR$ is given by 
    \begin{equation*}
        \Omega_\mR = 
        \begin{pmatrix}
            x^1_1 & -y^1_1 & \cdots & x^1_r & -y^1_r \\
            y^1_1 & x^1_1 & \cdots & y^1_r & x^1_r \\
            \vdots & \vdots &       & \vdots & \vdots \\
            x^r_1 & -y^r_1 & \cdots & x^r_r & -y^r_r \\
            y^r_1 & x^r_1 & \cdots & y^r_r & x^r_r
        \end{pmatrix}
        \overset{\text{similar to}}{\rightsquigarrow} 
        \begin{pmatrix}
            x^1_1 & \cdots & x^1_r & -y^1_1 & \cdots & -y^1_r \\ 
            \vdots & & \vdots & \vdots & & \vdots \\ 
            x^r_1 & \cdots & x^r_r & -y^r_1 & \cdots & -y^r_r \\ 
            y^1_1 & \cdots & y^1_r & x^1_1 & \cdots & x^1_r \\ 
            \vdots & & \vdots & \vdots & & \vdots \\ 
            y^r_1 & \cdots & y^r_r & x^r_1 & \cdots & x^r_r 
        \end{pmatrix} 
        = 
        \begin{pmatrix}
            A & -B \\
            B & A
        \end{pmatrix}
    \end{equation*}
    So
    \begin{align*}
        \det\left(\frac{1}{2\pi}\Omega_\mR\right) 
        &= \det\left(\frac{1}{2\pi}\begin{pmatrix} A & -B \\ B & A \end{pmatrix}\right) 
        = \frac{1}{(2\pi)^{2r}}\big|\det(A+\mi B)\big|^2 \\
        &= \left|\det\left(\frac{1}{2\pi}\Omega\right)\right|^2 
        = \left|\det\left(\frac{\mi}{2\pi}\Omega\right)\right|^2 
        = \left(\det\left(\frac{\mi}{2\pi}\Omega\right)\right)^2.
    \end{align*}
    The last equality holds since $\det\left(\frac{\mi}{2\pi}\Omega\right)$ is a real form. 
    Therefore, 
    \begin{equation*}
        \Pf\left(\frac{1}{2\pi}(R^\nabla)_\mR\right) = \pm\det\left(\frac{\mi}{2\pi}R^\nabla\right). 
    \end{equation*}
    To determine the sign, set $\Omega = -\mi I_r$, then 
    \begin{equation*}
        \Omega_\mR = 
        \begin{pmatrix}
            0 & 1 & & & \\ 
            -1 & 0 & & & \\ 
             & & \ddots & & \\
             & & & 0 & 1 \\
             & & & -1 & 0 
        \end{pmatrix} = J_{2r}
    \end{equation*} 
    So 
    \begin{equation*}
        \det(\mi\Omega) = \det(I_r) = 1 = \Pf(J_{2r}) = \Pf(\Omega_\mR).
    \end{equation*}
    The sign under this setting is $+1$, thus 
    \begin{equation*}
        \Pf\left(\frac{1}{2\pi}(R^\nabla)_\mR\right) = \det\left(\frac{\mi}{2\pi}R^\nabla\right)
        \implies e(E_\mR) = c_r(E). \qedhere
    \end{equation*}
\end{proof}

In fact, there is a more general Chern-Weil theory for principal $G$-bundles.

Let $\pi:P\to M$ be a principal $G$-bundle, $\mathfrak{g}$ be the Lie algebra of $G$. 
$\omega$ is a connection form on $P$, $\Omega$ is the curvature form. Recall that 
\begin{itemize}[leftmargin=*]
    \item $\omega\in\Omega^1(P,\mathfrak{g})$, $r_g^*\omega = \mathrm{Ad}_{g^{-1}}\omega$, and $\omega(\underline{A}) = A$ for all $A\in\mathfrak{g}$; 
    \item $\Omega\in\Omega^2(P,\mathfrak{g})$, $r_g^*\Omega = \mathrm{Ad}_{g^{-1}}\Omega$, and $\Omega(X,Y) = 0$ if $X$ or $Y$ is vertical. 
    % So $\Omega$ is a tensorial form of type $\mathrm{Ad}$; 
    \item There is a one-to-one correspondence between $\Omega^k_{\rho}(P,V)$ and $\Omega^k(M,P\times_{\rho}V)$:
    \begin{equation*}
        \left\{
            \begin{array}{c}
                k\text{-forms on }P\text{ valued in }V\text{ that are } \\ 
                G\text{-equivariant and horizontal} 
            \end{array}
        \right\}
        \overset{1-1}{\longleftrightarrow}
        \left\{
            \begin{array}{c}
                k\text{-forms on }M\text{ valued in }P\times_{\rho}V
            \end{array}
        \right\}
    \end{equation*}
    \item There is a one-to-one correspondence between $\Omega^k_{\text{basic}}(P,V)$ and $\Omega^k(M,V)$:
    \begin{align*}
        \left\{
            \begin{array}{c}
                k\text{-forms on }P\text{ valued in }V\text{ that are } \\ 
                G\text{-invariant and horizontal} 
            \end{array}
        \right\}
        \overset{1-1}{\longleftrightarrow}&
        \left\{
            \begin{array}{c}
                k\text{-forms on }M\text{ valued in }V
            \end{array}
        \right\} \\ 
        \pi^*\phi\; \reflectbox{$\longmapsto$}& \;\phi.
    \end{align*}
    \item The covariant derivative $D:\Omega^k(P,V)\to\Omega^{k+1}(P,V)$ 
    is defined by $D\varphi = (\md\varphi)^h$. In particular, 
    it maps $\Omega^k_{\rho}(P,V)$ to $\Omega^{k+1}_{\rho}(P,V)$; 
    \item The Bianchi identity: $D\Omega = \md\Omega + [\omega,\Omega] = 0$; 
    \item If $\varphi\in\Omega^k_{\rho}(P,V)$, then
    \begin{equation*}
        D\varphi = \md\varphi + \omega\cdot\varphi.
    \end{equation*}
    In particular, if $\varphi\in\Omega^k_{\text{basic}}(P,V)$, then $D\varphi = \md\varphi$.

\begin{definition}[Invariant polynomials]
    Let $G$ be a Lie group, $\mathfrak{g}$ be its Lie algebra. 
    A polynomial of degree $k$ on $\mathfrak{g}$ is an element of $\mathrm{Sym}^k(\mathfrak{g}^*)$. 
    We call a polynomial $\Phi\in\mathrm{Sym}^k(\mathfrak{g}^*)$ \emph{\textbf{$\mathrm{Ad}(G)$-invariant}} if 
    \begin{equation*}
        \Phi(\mathrm{Ad}_g X) = \Phi(X)\qquad\forall\,g\in G,\,X\in\mathfrak{g}.
    \end{equation*}
    The set of all $\mathrm{Ad}(G)$-invariant polynomials on $\mathfrak{g}$ is denoted by $\mathrm{Inv}(\mathfrak{g})$.
\end{definition}
\begin{theorem}
    For any invariant polynomial $\Phi\in\mathrm{Inv}(\mathfrak{g})$, we have
    \begin{itemize}[leftmargin=*]
        \item $\Phi(\Omega)$ is a basic form, thus defines a form $\Lambda$ on $M$ such that $\pi^*\Lambda = \Phi(\Omega)$; 
        \item $\Phi(\Omega)$ is closed, so as $\Lambda$; 
        \item the cohomology class $[\Lambda]\in H^{2k}(M)$ is independent of the choice of connection $\omega$ on $P$. 
    \end{itemize}
    The cohomology class $[\Lambda]$ is called the \emph{\textbf{characteristic class}} of $P$ associated to $\Phi$. The map 
    \begin{align*}
        c:\mathrm{Inv}(\mathfrak{g}) &\to H^{*}(M) \\
        \Phi &\mapsto [\Lambda],
    \end{align*}
    is called the \emph{\textbf{Chern-Weil homomorphism}}.
\end{theorem}
\begin{proof}
    \noindent\emph{\textbf{Step 1: $\Phi(\Omega)$ is a basic form.}}
    Since $\Omega$ is a tensorial form of type $\mathrm{Ad}$, we have
    \begin{equation*}
        r_g^*\Phi(\Omega) = \Phi(r_g^*\Omega) = \Phi(\mathrm{Ad}_{g^{-1}}\Omega) = \Phi(\Omega),
    \end{equation*}
    so $\Phi(\Omega)$ is $G$-invariant. Clearly, $\Phi(\Omega)$ is horizontal. 
    Thus $\Phi(\Omega)$ is a basic form and $\Phi(\Omega) = \pi^*\Lambda$ for some form $\Lambda$ on $M$.

    \noindent\emph{\textbf{Step 2: $\Phi(\Omega)$ is closed.}}
    Since $\Phi(\Omega)$ is basic, 
    \begin{equation*}
        \md\Phi(\Omega) = D\Phi(\Omega) = \Phi(D\Omega) = 0.
    \end{equation*}
    For the closedness of $\Lambda$, 
    \begin{equation*}
        \pi^*(\md\Lambda) = \md(\pi^*\Lambda) = \md\Phi(\Omega) = 0.
    \end{equation*}
    Thus $\md\Lambda = 0$ by the injectivity of $\pi^*$. 

    \noindent\emph{\textbf{Step 3: Independence of $\omega$.}}
    Let $\omega_0$ and $\omega_1$ be two connections on $P$, 
    $\Omega_0$ and $\Omega_1$ be the curvature forms,
    $\Lambda_0$ and $\Lambda_1$ are forms on $M$ such that 
    $\pi^*\Lambda_i = \Phi(\Omega_i)$, $i=0,1$. 

    Consider the family of connections $\omega_t = (1-t)\omega_0 + t\omega_1$,
    and the corresponding curvature forms $\Omega_t$. 
    % Then $\omega_t$ is a connection on $P$ for each $t\in[0,1]$, 
    
\end{proof}
