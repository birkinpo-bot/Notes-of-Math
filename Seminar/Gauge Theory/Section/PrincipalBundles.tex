\section{Principal Bundles and Connections}
\subsection{Principal Bundles}
Principal bundles are both a special type of fiber bundle and the most important one. 
In the following, we will introduce the definition of principal bundles and some basic structures on them. 
\begin{definition}[$G$-action]
    A \textbf{smooth right action} of a Lie group $G$ on a manifold $P$ is a smooth map 
    \begin{equation*}
        \mu: M\times G \to M, 
    \end{equation*}
    denoted by $\mu(x,g):=x\cdot g$, such that for all $x\in M$ and $g,h\in G$, 
    \begin{equation*}
        x\cdot e = x, \quad (x\cdot g)\cdot h = x\cdot (gh).
    \end{equation*}
    where $e$ is the identity element of $G$. 
    A manifold $M$ with a smooth right $G$-action is called a \textbf{{\rm(}right{\rm)} $G$-manifold}.

    We say the action is \textbf{free} if for all $x\in M$, $x\cdot g = x$ implies $g=e$.
\end{definition}
\begin{definition}[$G$-equivariant map]
    Let $M$ and $N$ be two $G$-manifolds. 
    A smooth map $f: M\to N$ is called \textbf{$G$-equivariant} if for all $x\in M$ and $g\in G$, 
    \begin{equation*}
        f(x\cdot g) = f(x)\cdot g.
    \end{equation*}
\end{definition}
\begin{definition}[Principal $G$-bundle]
    A \textbf{principal $G$-bundle} is a fiber bundle $\pi: P\to M$ with fiber $G$ 
    together with a free right $G$-action on $P$ such that for each $x\in M$, 
    the local trivialization 
    \begin{equation*}
        \phi_U: \pi^{-1}(U)\to U\times G,
    \end{equation*}
    is $G$-equivariant, where the right $G$-action on $U\times G$ is given by 
    \begin{equation*}
        (x,h)\cdot g = (x,hg), \quad \forall x\in U, g,h\in G.
    \end{equation*}    
\end{definition}
By a \textbf{local trivialization} we mean a open neighborhood $U$ of $x$ and a diffeomorphism 
$\phi_U: \pi^{-1}(U)\to U\times G$ satisfying the following commutative diagram:
\begin{center}
    \begin{tikzcd}
        \pi^{-1}(U) \arrow[rr, "\phi_U"] \arrow[dr, "\pi"'] & & U\times G \arrow[dl, "\mathrm{pr}_1"] \\
        & U &
    \end{tikzcd}
\end{center}
where $\mathrm{pr}_1: U\times G\to U$ is the projection map.
\begin{example}
    Let $G$ be a Lie group and $M$ be a smooth manifold, then the trivial bundle 
    $p_1: M\times G\to M$ is a principal $G$-bundle over $M$.
\end{example}
\begin{example}
    If $G$ is a finite group equipped with the discrete topology, then a principal $G$-bundle is exactly a covering space.
\end{example}
\begin{example}
    Let $G$ be a Lie group and $H$ be a closed subgroup of $G$. 
    Then the quotient space $G/H$ is a smooth manifold and the projection map $\pi: G\to G/H$ 
    is a principal $H$-bundle over $G/H$. For example, the Hopf fibration $S^1\to S^3\to S^2$. 
\end{example}
\begin{example}
    Let $\pi:E\to M$ be a rank $m$ real vector bundle over $M$. 
    Then the frame bundle $\mathrm{Fr}(E)$ is a principal $\GL(m,\mR)$-bundle over $M$. 
    If $E$ is a complex vector bundle, then $\mathrm{Fr}(E)$ is a principal $\GL(m,\mC)$-bundle over $M$. 
\end{example}
\begin{proposition}
    Let $\pi: P\to M$ be a principal $G$-bundle. 
    Then the right $G$-action on $P$ is transitive on each fiber.
\end{proposition}
\begin{proof}
    Since $G$ acts transitively on $\{x\}\times G$ and the fiber diffeomorphism $\phi_{U,x}:P_x\to G$ 
    is $G$-equivariant, thus $G$ acts transitively on $P_x$.
\end{proof} 
\begin{lemma}\label{lem:equivariant-maps-are-left-translations}
    For any group $G$, all right $G$-equivariant maps are exactly the left translations. 
\end{lemma}
\begin{proof}
    Let $f: G\to G$ be a right $G$-equivariant map, then for all $g\in G$, 
    \begin{equation*}
        f(g) = f(e\cdot g) = f(e)\cdot g = l_{f(e)}(g),
    \end{equation*}
    where $l_{f(e)}: G\to G$ is the left translation by $f(e)$.
\end{proof}
\begin{definition}[Cocycle definition]
    A \textbf{principal $G$-bundle} is a fiber bundle $\pi: P\to M$ with fiber $G$ 
    together with a local trivialization $\{(U_\alpha, \phi_\alpha)\}$ 
    such that for each $U_{\alpha\beta}:=U_\alpha\cap U_\beta\neq\emptyset$, 
    there exists a smooth map $g_{\alpha\beta}: U_{\alpha\beta}\to G$ such that 
    \begin{equation*}
        \phi_\alpha\circ\phi_\beta^{-1}(x,h) = (x, g_{\alpha\beta}(x)h), 
        \quad \forall (x,h)\in U_{\alpha\beta}\times G.
    \end{equation*}
    In addition, the transition functions $\{g_{\alpha\beta}\}$ satisfy the following cocycle condition: 
    \begin{gather*}
        % g_{\alpha\alpha}(x) = e, \quad \forall x\in U_\alpha, \\
        g_{\alpha\beta}(x)g_{\beta\gamma}(x) = g_{\alpha\gamma}(x), 
        \quad \forall x\in U_\alpha\cap U_\beta\cap U_\gamma\neq\emptyset.
    \end{gather*}
\end{definition}
\begin{remark}
    By Lemma \ref{lem:equivariant-maps-are-left-translations}, 
    one can easily check that the two definitions of principal bundles are equivalent. 
    These two definitions characterize $G$-principal bundles from different perspectives.
\end{remark}
\subsubsection*{Fundamental Vector Fields}
Suppose $P$ is a $G$-manifold, then for each $A\in\mathfrak{g}$, we can define a vector field $\underline{A}$ on $P$ by
\begin{equation*}
    \underline{A}_p := \left.\frac{d}{dt}\right|_{t=0} p\cdot\exp(tA), \quad \forall p\in P.
\end{equation*}
The vector field $\underline{A}$ is called the \textbf{fundamental vector field} associated to $A$.
\begin{fact}
    For each $A\in\mathfrak{g}$, $\underline{A}$ is a $C^\infty$ vector field on $P$.
\end{fact} 
The fundamental vector field construction gives rise to a map 
\begin{equation*}
    \sigma: \mathfrak{g}\to\mathfrak{X}(P),\footnotemark \quad \sigma(A):=\underline{A}.
\end{equation*}
\footnotetext{$\mathfrak{X}(P):=C^\infty(P,TP)$ is the space of smooth vector fields on $P$.}
For $p\in P$, define $j_p: G\to P$ by $j_p(g) := p\cdot g$, then $j_p{}_*:=j_p{}_*{}_e$ is 
\begin{equation*}
    j_p{}_*(A) = \left.\frac{\md}{\md t}\right|_{t=0} j_p(\exp(tA)) 
    = \left.\frac{\md}{\md t}\right|_{t=0} p\cdot\exp(tA) = \underline{A}_p.
\end{equation*}
This gives another discription of the fundamental vector field: $\underline{A}_p = j_p{}_*(A)$. 
It also implies that the map $\sigma: \mathfrak{g}\to\mathfrak{X}(P)$ is $\mR$-linear. In fact, we have 
\begin{fact}
    The map $\sigma: \mathfrak{g}\to\mathfrak{X}(P)$ is a Lie algebra homomorphism, i.e., 
    \begin{equation*}
        [\underline{A}, \underline{B}] = \underline{[A,B]}, \quad \forall A,B\in\mathfrak{g}.
    \end{equation*}
\end{fact}
\begin{fact}
    Let $P$ be a $G$-manifold, then the fundamental vector fields $\underline{A}$ vanishes at $p$ 
    if and only if $A\in\mathrm{Lie}(\mathrm{Stab}(p))$. Thus for any $p\in P$, the kernel of 
    $j_p{}_*:\mathfrak{g}\to T_pP$ is exactly $\mathrm{Lie}(\mathrm{Stab}(p))$. In particular, 
    if the $G$-action on $P$ is free, then $j_p{}_*$ is injective for all $p\in P$.
\end{fact}
\begin{example}
    Let $G$ acts on itself by right translation, 
    then the fundamental vector field $\underline{A}$ is exactly the left-invariant vector field generated by $A$.
\end{example}
For $g\in G$, let $c_g: G\to G$ be the conjugation map $c_g(h) := ghg^{-1}$. 
Then the differential of $c_g$ at the identity element $e$ is the adjoint representation 
$\mathrm{Ad}_g:=(c_g)_*: \mathfrak{g}\to\mathfrak{g}$. 
\begin{proposition}
    For each $A\in\mathfrak{g}$ and $g\in G$, we have 
    \begin{equation*}
        (r_g)_*\underline{A} = \underline{\mathrm{Ad}_{g^{-1}}(A)}.
    \end{equation*}
    where $r_g: P\to P$ is the right translation by $g$, i.e., $r_g(p) := p\cdot g$.
\end{proposition}
\begin{proof}
    For each $p\in P$ and $h\in G$, we have 
    \begin{equation*}
        (r_g\circ j_p)(h) = r_g(p\cdot h) = (p\cdot h)\cdot g = pg\cdot (g^{-1}hg) = (j_{pg}\circ c_{g^{-1}})(h).
    \end{equation*}
    By the chain rule, 
    \begin{equation*}
        (r_g)_*(\underline{A}_p) = (r_g)_*j_p{}_*(A) = j_{pg}{}_*(c_{g^{-1}})_*(A) 
        = j_{pg}{}_*(\mathrm{Ad}_{g^{-1}}(A))
        = \underline{\mathrm{Ad}_{g^{-1}}(A)}_{pg}.\qedhere
    \end{equation*}
\end{proof}


\subsubsection{Vertical Subbundle of $TP$}
Let $\pi: P\to M$ be a principal $G$-bundle, at each $p\in P$, the differential $\pi_*{}_p: T_pP\to T_{\pi(p)}M$ is 
surjective. We define $\mathcal{V}_p := \ker \pi_*{}_p\subset T_pP$ to be the \textbf{vertical tangent space} at $p$. 
Then $\mathcal{V} := \bigsqcup_{p\in P} \mathcal{V}_p$ is a subbundle of $TP$, called the \textbf{vertical subbundle} of $TP$. 
An element of $\mathcal{V}_p$ is called a \textbf{vertical tangent vector} at $p$. 

We have a short exact sequence of vector spaces
\begin{center}
    \begin{tikzcd}
        0 \arrow[r] & {\mathcal{V}_p} \arrow[r, hook] & T_pP \arrow[r, "\pi_*{}_p"] & {T_{\pi(p)}M} \arrow[r] & 0
    \end{tikzcd}
\end{center}
and
\begin{equation*}
     \dim \mathcal{V}_p = \dim T_p P - \dim T_{\pi(p)} M = \dim G.
\end{equation*}

\begin{proposition}
    For any $A\in\mathfrak{g}$, the fundamental vector field $\underline{A}$ is vertical at every point $p\in P$.
\end{proposition}
\begin{proof}
    Since $(\pi\circ j_p)(g) = \pi(p\cdot g) = \pi(p)$ is a constant map for all $g\in G$, 
    we have $\pi_*{}_p(\underline{A}_p) = \pi_*{}_p j_p{}_*(A) = (\pi\circ j_p)_*(A)= 0$.
\end{proof}
This proposition implies that $j_p{}_*(\mathfrak{g})\subset \mathcal{V}_p$ for all $p\in P$. 
In fact, we have the following stronger result.
\begin{proposition}
    For each $p\in P$, the map $j_p{}_*: \mathfrak{g} \to \mathcal{V}_p$ is an isomorphism.
\end{proposition}
\begin{proof}
    Since $G$ acts freely on $P$, the map $j_p{}_*$ is injective. On the other hand, 
    $\dim \mathcal{V}_p = \dim G = \dim \mathfrak{g}$, thus $j_p{}_*$ is an isomorphism.
\end{proof}
\begin{corollary}
    The vertical vectors at $p\in P$ are exactly the fundamental vector at $p$. Moreover, 
    the vertical subbundle $\mathcal{V}$ of $TP$ is isomorphic to the trivial bundle $P\times\mathfrak{g}$.
\end{corollary}
\begin{proof}
    The first statement follows from the previous proposition. For the second statement, 
    let $\ft{A}{1}{l}$ be a basis of $\mathfrak{g}$, where $l = \dim G$, 
    then $\underline{A}_1, \ldots, \underline{A}_l$ is a global frame of $\mathcal{V}$, 
    thus $\mathcal{V}$ is a trivial bundle. 
\end{proof}
The map $\pi_*: TP\to TM$ induces a bundle map $\tilde{\pi}_*: TP\to \pi^*TM$ over $P$. 
Then we have a short exact sequence of vector bundles over $P$: 
\begin{center}
    \begin{tikzcd}
        0 \arrow[r] & {\mathcal{V}} \arrow[r, hook] & TP \arrow[r, "\tilde{\pi}_*"] & {\pi^*TM} \arrow[r] & 0
    \end{tikzcd}
\end{center}
\subsection{Connections on Principal Bundles}
\subsubsection{Horizontal Distributions}
In the category of smooth vector bundles, all short exact sequence of vector bundles splits. 
Corresponding to the short exact sequence appearing in the previous chapter, 
there exists a smooth subbundle $\mathcal{H}$ of $TP$ such that $TP = \mathcal{V}\oplus\mathcal{H}$.
We call $\mathcal{H}$ a \textbf{horizontal distribution} of $TP$. 

In general, there is no canonical choice of $\mathcal{H}$. 
A choice of $\mathcal{H}$ is the same as a splitting $k: \pi^*TM\to TP$ such that $\tilde{\pi}_*\circ k = \mathrm{id}_{\pi^*TM}$. 
\[\begin{tikzcd}
    0 & {\mathcal{V}} & TP & {\pi^*TM} & 0
    \arrow[from=1-1, to=1-2]
    \arrow[hook, from=1-2, to=1-3]
    \arrow["\tilde{\pi}_*", shift left, from=1-3, to=1-4]
    \arrow["k", shift left, from=1-4, to=1-3]
    \arrow[from=1-4, to=1-5]
\end{tikzcd}\]

We say $\mathcal{H}$ is \textbf{right-invariant} if for all $g\in G$ and $p\in P$, 
$r_g{}_*{}_p(\mathcal{H}_p) = \mathcal{H}_{pg}$, where $r_g: P\to P,\, r_g(p) = p\cdot g$. 

\subsubsection{Ehresmann Connections}
Let $\pi: P\to M$ be a principal $G$-bundle, $\mathcal{H}$ be a horizontal distribution of $TP$. 
For each $p\in P$, we define a projection map $v: T_pP=\mathcal{V}_p\oplus\mathcal{H}_p\to \mathcal{V}_p$. 
For $Y_p\in T_pP$, we call $v(Y_p)$ the \textbf{vertical component} of $Y_p$.\footnote{Although $\mathcal{V}_p$ 
is intrinsically defined, the vertical component of a tangent vector depends on the choice of $\mathcal{H}_p$.} 

Let $\iota: \mathcal{V}\hookrightarrow TP$ be the inclusion map, then $v\circ\iota = \mathrm{id}_{\mathcal{V}}$.
We actually obtain another splitting of the short exact sequence: 
\[\begin{tikzcd}
    0 & {\mathcal{V}} & TP & {\pi^*TM} & 0
    \arrow[from=1-1, to=1-2]
    \arrow["\iota", hook, shift left, from=1-2, to=1-3]
    \arrow["\tilde{\pi}_*", from=1-3, to=1-4]
    \arrow["v", shift left, from=1-3, to=1-2]
    \arrow[from=1-4, to=1-5]
\end{tikzcd}\]
Recall that for each $p\in P$, the map $j_p: G\to P$ defined by $j_p(g) = p\cdot g$. 
We define a $\mathfrak{g}$-valued 1-form $\omega$ on $P$ by
\begin{equation*}
    \omega_p:= j_{p*}^{-1}\circ v: T_pP\xrightarrow{v} \mathcal{V}_p\xrightarrow{j_{p*}^{-1}} \mathfrak{g}.
\end{equation*}
In terms of $\omega$, the vertical component of $Y_p\in T_pP$ is given by 
\begin{equation*}
    v(Y_p) = j_{p*}(\omega_p(Y_p)).
\end{equation*}
In summary, given a projection $v: TP\to \mathcal{V}$, we induce the following composition of maps:
\begin{equation*}
    \omega:TP \xrightarrow{v} \mathcal{V} \xrightarrow{j_*^{-1}} P\times\mathfrak{g} \xrightarrow{\mathrm{pr}_2} \mathfrak{g}
    \quad\rightsquigarrow\quad \omega\in\Omega^1(P,\mathfrak{g}).
\end{equation*}
% \[\begin{tikzcd}
%     TP \arrow[r, "v"] & \mathcal{V} \arrow[r, "j_*^{-1}"] & P\times\mathfrak{g} \arrow[r, "\mathrm{pr}_2"] & \mathfrak{g}
% \end{tikzcd}\]
So the information of the horizontal distribution $\mathcal{H}$ is encoded in $\omega$.

\begin{theorem}
    If $\mathcal{H}$ is a smooth, right-invariant horizontal distribution, 
    then the $\mathfrak{g}$-valued 1-form $\omega$ defined above 
    satisfies the following properties:
    \begin{enumerate}
        \item $\omega_p(\underline{A}_p) = A,\;\forall\,p\in P,\,A\in\mathfrak{g}$.
        \item {\rm(}$G$-equivalence{\rm)} $(r_g)^*\omega = (\mathrm{Ad}_{g^{-1}})\omega,\;\forall\,g\in G$.
        \item $\omega$ is smooth.
    \end{enumerate}

    We say a $\mathfrak{g}$-valued 1-form $\omega$ on $P$ satisfying the above properties is 
    an \textbf{Ehresmann connection} or simply a \textbf{connection} on $P$.
\end{theorem}
\begin{proof}
    $\textit{1}$. Since $\underline{A}$ is already vertical, 
    \begin{equation*}
        \omega_p(\underline{A}_p) = j_{p*}^{-1}(v(\underline{A}_p)) = j_{p*}^{-1}(\underline{A}_p) = A.
    \end{equation*}

    \noindent$\textit{2}$. For each $p\in P$ and $Y_p\in T_pP$, we need to show 
    \begin{equation*}
        \omega_{pg}((r_g)_*Y_p) = (\mathrm{Ad}_{g^{-1}})\omega_p(Y_p).
    \end{equation*}
    Since both side are linear in $Y_p$, we only need to check the above equality for 
    $Y_p\in \mathcal{V}_p$ and $Y_p\in \mathcal{H}_p$ respectively.

    If $Y_p\in \mathcal{V}_p$, then $Y_p = \underline{A}_p$ for some $A\in\mathfrak{g}$, thus
    \begin{equation*}
        \omega_{pg}(r_g{}_*\underline{A}_p) 
        = \omega_{pg}(\underline{\mathrm{Ad}_{g^{-1}}(A)}_{pg}) 
        = \mathrm{Ad}_{g^{-1}}(A) = (\mathrm{Ad}_{g^{-1}})\omega_p(\underline{A}_p).
    \end{equation*}

    If $Y_p\in \mathcal{H}_p$, by the right-invariance of $\mathcal{H}$, we have $r_g{}_*(Y_p)\in \mathcal{H}_{pg}$, thus 
    \begin{equation*}
        \omega_{pg}(r_g{}_*Y_p) = 0 = (\mathrm{Ad}_{g^{-1}})\omega_p(Y_p).
    \end{equation*}

    \noindent$\textit{3}$. Omit.
\end{proof}

Viewing $\omega$ as a map $\omega: TP\to \mathfrak{g}$. Both $TP$ and $\mathfrak{g}$ are $G$-manifolds, 
where
\begin{equation*}
    g\cdot Y_p := (r_g)_*Y_p, \quad g\cdot A := \mathrm{Ad}_{g^{-1}}(A). 
\end{equation*}
Then the $G$-equivalence condition of $\omega$ 
\begin{equation*}
    \omega_{pg}((r_g)_*Y_p) = (\mathrm{Ad}_{g^{-1}})\omega_p(Y_p),
\end{equation*}
can be rephrased as $\omega: TP\to \mathfrak{g}$ is a $G$-equivariant map.

\subsubsection{The Horizontal Distribution of an Ehresmann Connection}
In the previous section we have shown that a smooth, right-invariant horizontal distribution $\mathcal{H}$ on $P$ induces an Ehresmann connection $\omega$. 
We now prove the converse. 
\begin{theorem}
    If $\omega$ is an Ehresmann connection on the principal $G$-bundle $\pi: P\to M$, 
    then $\mathcal{H}_p := \ker \omega_p,\,p\in P$ defines a smooth, right-invariant horizontal distribution on $P$.
\end{theorem}
\begin{proof}
    We need to verify three properties: 

    \noindent$\textit{1}$. For each $p\in P$, there is an exact sequence 
    \begin{center}
        \begin{tikzcd}
            0 \arrow[r] & {\mathcal{H}_p} \arrow[r, hook] & T_pP \arrow[r, "\omega_p"] & {\mathfrak{g}} \arrow[r] & 0
        \end{tikzcd}
    \end{center}
    The map $j_p{}_*: \mathfrak{g}\to\mathcal{V}_p\subset T_pP$ provides a splitting of the above sequence, 
    thus 
    \begin{equation*}
        T_pP \cong \mathfrak{g}\oplus\mathcal{H}_p\cong \mathcal{V}_p\oplus\mathcal{H}_p.
    \end{equation*}

    \noindent$\textit{2}$. Suppose $Y_p\in \mathcal{H}_p$, then $\omega_p(Y_p) = 0$. By the $G$-equivalence of $\omega$, we have
    \begin{equation*}
        \omega_{pg}(r_g{}_*Y_p) = (r_g^*\omega)_p(Y_p) = (\mathrm{Ad}_{g^{-1}})\omega_p(Y_p) = 0.
    \end{equation*}
    Hence $r_g{}_*Y_p\in \mathcal{H}_{pg}$.

    \noindent$\textit{3}$. $\mathcal{H}$ is a smooth distribution. We omit the details here.
\end{proof}
